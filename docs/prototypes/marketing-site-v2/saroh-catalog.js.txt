/* Saroh plans and modules catalog, shared by the Admin Console (Plans & modules),
   the public pricing page and the merchant dashboard. Everything lives in
   localStorage so all three agree in the prototype.
   - versions: every publish is a version. Businesses sit on a version; "keep"
     leaves them on theirs, "move" schedules a switch on their next billing date
     (at least 7 days' notice).
   - draft: edits in the admin console before publishing.
   - biz: which plan and version each business is on, plus a scheduled move.
   - overrides: per-business changes that last until removed. */
(function () {
  const K = { versions: "saroh.catalog.versions", draft: "saroh.catalog.draft", biz: "saroh.catalog.biz", ovr: "saroh.catalog.overrides", coupons: "saroh.catalog.coupons", bill: "saroh.catalog.billing" };
  const C = (text, card, limit, per) => ({ inc: true, text: text, card: card || "", limit: limit == null ? null : limit, per: per || "" });
  const X = (off) => ({ inc: false, off: off || "locked" });
  const DEFAULT = {
    plans: [
      { id: "free", name: "Free", price: 0, tagline: "For getting your business online.", cta: "Start free", featured: false, retired: false },
      { id: "grow", name: "Grow", price: 1000, tagline: "For a business that sells, books and bills.", cta: "Choose Grow", featured: true, retired: false },
      { id: "pro", name: "Pro", price: 5000, tagline: "For your own look and a bigger team.", cta: "Choose Pro", featured: false, retired: false }
    ],
    groups: [{ id: "site", name: "Your site" }, { id: "selling", name: "Selling" }, { id: "money", name: "Money" }, { id: "team", name: "Team" }, { id: "connect", name: "Connections" }],
    modules: [
      { id: "website", name: "Website", group: "site", pricing: "show", menu: "website", what: "Your site, with its pages and booking page.", cells: { free: C("1 site", "One website"), grow: C("1 site on your own domain", "Your website on your own domain"), pro: C("1 site on your own domain", "Your website on your own domain") } },
      { id: "themes", name: "Themes, fonts and templates", group: "site", pricing: "show", what: "Change the site's theme and fonts, or pick another template.", cells: { free: X("hidden"), grow: X("hidden"), pro: C("Change the theme and fonts, or pick another template", "Change your site's theme and fonts, or pick another template") } },
      { id: "review", name: "Review changes with your team", group: "site", pricing: "show", what: "Changes to the site wait for a teammate's review before they go live.", cells: { free: X("hidden"), grow: X("hidden"), pro: C("Included", "Review changes with your team before they go live") } },
      { id: "blog", name: "Blog posts", group: "site", pricing: "show", what: "Posts on your site's journal.", cells: { free: C("5", "5 blog posts", 5), grow: C("50", "50 blog posts", 50), pro: C("Included", "Blog posts, no limit") } },
      { id: "products", name: "Products", group: "selling", pricing: "show", menu: "sell", child: "Products", what: "Everything you sell, with sizes, prices and stock.", cells: { free: C("5", "5 products", 5), grow: C("100", "100 products", 100), pro: C("1,000", "1,000 products", 1000) } },
      { id: "orders", name: "Orders", group: "selling", pricing: "show", menu: "sell", child: "Orders", what: "Every order from your site and the counter, in one list.", cells: { free: C("Up to 10 a month, paid offline", "Up to 10 orders a month, paid offline", 10, "month"), grow: C("Included", "Orders, paid online or offline"), pro: C("Included", "Orders, paid online or offline") } },
      { id: "subscriptions", name: "Subscriptions", group: "selling", pricing: "show", what: "Plans and memberships that renew themselves.", cells: { free: X("locked"), grow: C("Included", "Subscriptions"), pro: C("Included", "Subscriptions") } },
      { id: "bookings", name: "Bookings and appointments", group: "selling", pricing: "show", menu: "bookingsx", what: "Classes, sessions and appointments customers book themselves.", cells: { free: C("Up to 10 a month", "Up to 10 bookings and appointments a month", 10, "month"), grow: C("Included", "Bookings and appointments"), pro: C("Included", "Bookings and appointments") } },
      { id: "invoicing", name: "Billing and invoicing", group: "money", pricing: "show", menu: "paymentsx", what: "Payments, GST invoices and bills of supply, made for you.", cells: { free: X("locked"), grow: C("Included", "Billing and invoicing"), pro: C("Included", "Billing and invoicing") } },
      { id: "members", name: "Team members", group: "team", pricing: "show", what: "People who can sign in to your dashboard.", cells: { free: C("1", "1 team member", 1), grow: C("3", "3 team members", 3), pro: C("10", "10 team members", 10) } },
      { id: "roles", name: "Custom roles", group: "team", pricing: "show", what: "Roles you define, beyond Owner, Admin, Member and Reviewer.", cells: { free: X("hidden"), grow: X("hidden"), pro: C("Included", "Custom roles") } },
      { id: "integrations", name: "Third-party integrations", group: "connect", pricing: "show", what: "Connections to other tools.", cells: { free: X("hidden"), grow: C("Up to 5", "Up to 5 third-party connections", 5), pro: C("Up to 50", "Up to 50 third-party connections", 50) } }
    ]
  };
  const BIZ_DEFAULT = {
    rye: { plan: "grow", v: 1 }, pulse: { plan: "pro", v: 1 }, dent: { plan: "grow", v: 1 }, lumen: { plan: "grow", v: 1 },
    iron: { plan: "free", v: 1 }, meridian: { plan: "free", v: 1 }, northwind: { plan: "free", v: 1 }, kettle: { plan: "free", v: 1 },
    prana: { plan: "free", v: 1 }, carepoint: { plan: "pro", v: 1 }
  };
  const clone = (x) => JSON.parse(JSON.stringify(x));
  const read = (k, d) => { try { const v = localStorage.getItem(k); return v ? JSON.parse(v) : d; } catch (e) { return d; } };
  const write = (k, v) => { try { if (v == null) localStorage.removeItem(k); else localStorage.setItem(k, JSON.stringify(v)); } catch (e) {} emit(); };
  const subs = new Set();
  function emit() { subs.forEach((f) => { try { f(); } catch (e) {} }); }
  if (typeof window !== "undefined") window.addEventListener("storage", (e) => { if (e.key && e.key.indexOf("saroh.catalog.") === 0) emit(); });
  const fmtDate = (d) => d.getDate() + " " + ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"][d.getMonth()] + " " + d.getFullYear();

  function versions() { return read(K.versions, null) || [{ v: 1, at: "26 Sep 2026", by: "Saroh team", note: "First published pricing: Free, Grow and Pro.", policy: "", catalog: DEFAULT }]; }
  /* A version can be scheduled (goLive in the future). Live is the newest version whose goLive has arrived. */
  const todayISO = () => { const d = new Date(); return d.getFullYear() + "-" + String(d.getMonth() + 1).padStart(2, "0") + "-" + String(d.getDate()).padStart(2, "0"); };
  function liveIdx(vs) { let i = 0; const t = todayISO(); vs.forEach((v, k) => { if (!v.goLive || v.goLive <= t) i = k; }); return i; }
  function live() { const vs = versions(); return vs[liveIdx(vs)].catalog; }
  function liveV() { const vs = versions(); return vs[liveIdx(vs)].v; }
  function scheduled() { const t = todayISO(); return versions().filter((v) => v.goLive && v.goLive > t); }
  function cancelScheduled(n) {
    const t = todayISO(); write(K.versions, versions().filter((v) => !(v.v === n && v.goLive && v.goLive > t)));
    const store = read(K.biz, {}); Object.keys(store).forEach((k) => { if (store[k].moving && store[k].moving.v === n) delete store[k].moving; }); write(K.biz, store);
  }
  /* First billing date (1st of a month) at least 7 days after a date. */
  function billingFrom(d0) { const t = new Date(d0.getFullYear(), d0.getMonth(), d0.getDate() + 7); return t.getDate() === 1 ? t : new Date(t.getFullYear(), t.getMonth() + 1, 1); }
  const COUPON_DEFAULT = [{ id: "c1", code: "LAUNCH500", off: 500, plans: ["grow", "pro"], months: 3, active: true, uses: 4 }];
  function coupons() { return read(K.coupons, null) || clone(COUPON_DEFAULT); }
  function saveCoupons(list) { write(K.coupons, list); }
  /* What a business has chosen for itself: monthly or yearly, add-ons bought (id -> how many), and an applied coupon. */
  function billing(id) { return Object.assign({ cycle: "month", addons: {}, coupon: null, used: [] }, read(K.bill, {})[id] || {}); }
  function setBilling(id, patch) { const all = read(K.bill, {}); all[id] = Object.assign(billing(id), patch); write(K.bill, all); }
  const ADDON_MOD = { members: "members", products: "products", orders: "orders", bookings: "bookings", integrations: "integrations" };
  function version(v) { return (versions().find((x) => x.v === v) || versions()[versions().length - 1]).catalog; }
  function draft() { return read(K.draft, null); }
  function editing() { return draft() || clone(live()); }
  function saveDraft(c) { write(K.draft, c); }
  function discardDraft() { write(K.draft, null); }
  function nextBilling() {
    const t = new Date(); let d = new Date(t.getFullYear(), t.getMonth() + 1, 1);
    if ((d - t) / 864e5 < 7) d = new Date(t.getFullYear(), t.getMonth() + 2, 1);
    return d;
  }
  function bizAll() { return Object.assign({}, BIZ_DEFAULT, read(K.biz, {})); }
  function bizRec(id) {
    const b = bizAll()[id] || { plan: "free", v: liveV() };
    if (b.moving && new Date(b.moving.onISO) <= new Date()) return { plan: b.plan, v: b.moving.v };
    return b;
  }
  function setBiz(id, patch) { const all = read(K.biz, {}); all[id] = Object.assign({}, bizAll()[id] || { plan: "free", v: liveV() }, patch); write(K.biz, all); }
  function onPlan(planId) { const all = bizAll(); return Object.keys(all).filter((k) => all[k].plan === planId); }

  /* What changed between two catalogs, in words, for the publish dialog and the log. */
  function diff(a, b) {
    const out = [], pn = (c, id) => ((c.plans.find((p) => p.id === id) || {}).name || id);
    b.plans.forEach((p) => {
      const o = a.plans.find((x) => x.id === p.id);
      if (!o) { out.push("New plan: " + p.name + " at ₹" + p.price.toLocaleString("en-IN")); return; }
      if (o.name !== p.name) out.push(o.name + " renamed to " + p.name);
      if (o.price !== p.price) out.push(p.name + ": ₹" + o.price.toLocaleString("en-IN") + " → ₹" + p.price.toLocaleString("en-IN") + " a month");
      if (!o.retired && p.retired) out.push(p.name + " retired: no new businesses can choose it");
      if (o.retired && !p.retired) out.push(p.name + " offered again");
      if (o.tagline !== p.tagline || o.cta !== p.cta || o.featured !== p.featured) out.push(p.name + ": card wording or highlight changed");
    });
    if (a.plans.map((p) => p.id).join() !== b.plans.map((p) => p.id).filter((id) => a.plans.some((x) => x.id === id)).join()) out.push("Plans reordered");
    b.modules.forEach((m) => {
      const o = a.modules.find((x) => x.id === m.id);
      if (!o) { out.push("New module: " + m.name + (m.pricing === "soon" ? " (coming soon)" : m.pricing === "hidden" ? " (not on the pricing page)" : "")); return; }
      if (o.name !== m.name) out.push(o.name + " renamed to " + m.name);
      if (o.pricing !== m.pricing) out.push(m.name + ": pricing page " + ({ show: "shown", soon: "coming soon", hidden: "hidden" })[m.pricing]);
      if (o.group !== m.group) out.push(m.name + " moved to " + ((b.groups.find((g) => g.id === m.group) || {}).name || m.group));
      b.plans.forEach((p) => {
        const x = o.cells[p.id] || X(), y = m.cells[p.id] || X();
        if (x.inc !== y.inc) out.push(m.name + (y.inc ? " added to " : " removed from ") + pn(b, p.id));
        else if (y.inc && (x.text !== y.text || x.limit !== y.limit || x.per !== y.per)) out.push(m.name + " on " + pn(b, p.id) + ": " + (x.text || "—") + " → " + (y.text || "—"));
        else if (!y.inc && x.off !== y.off) out.push(m.name + " on " + pn(b, p.id) + ": " + (y.off === "hidden" ? "hidden" : "shown locked") + " in the dashboard");
      });
    });
    a.modules.forEach((m) => { if (!b.modules.find((x) => x.id === m.id)) out.push("Module removed: " + m.name); });
    const ord = (c) => c.modules.map((m) => m.id).filter((id) => a.modules.some((x) => x.id === id) && b.modules.some((x) => x.id === id)).join();
    if (ord(a) !== ord(b)) out.push("Rows reordered on the pricing page");
    return out;
  }
  function publish(opts) {
    const d = draft(); if (!d) return null;
    const vs = versions(), prev = vs[vs.length - 1], v = prev.v + 1, changes = opts.changes || diff(live(), d);
    const go = opts.goLive ? new Date(opts.goLive + "T00:00:00") : null;
    const nb = go ? billingFrom(go) : nextBilling(), onISO = nb.getFullYear() + "-" + String(nb.getMonth() + 1).padStart(2, "0") + "-01", on = fmtDate(nb);
    vs.push({ v: v, at: fmtDate(new Date()), goLive: opts.goLive || "", goLiveAt: go ? fmtDate(go) : "", by: opts.by || "Saroh team", note: opts.note || "", policy: opts.policy, moveOn: opts.policy === "move" ? on : "", changes: changes, catalog: d });
    write(K.versions, vs);
    if (opts.policy === "move") { const all = bizAll(), store = read(K.biz, {}); Object.keys(all).forEach((k) => { if (all[k].v < v) store[k] = Object.assign({}, all[k], { moving: { v: v, on: on, onISO: onISO } }); }); write(K.biz, store); }
    write(K.draft, null);
    return v;
  }
  function overrides(id) { return (read(K.ovr, {})[id]) || []; }
  function addOverride(id, o) { const all = read(K.ovr, {}); all[id] = (all[id] || []).concat([Object.assign({ at: fmtDate(new Date()), by: "Saroh team" }, o)]); write(K.ovr, all); }
  function removeOverride(id, i) { const all = read(K.ovr, {}); all[id] = (all[id] || []).filter((_, j) => j !== i); write(K.ovr, all); }

  /* What one business gets for one module, after its plan, its version and its overrides. */
  function access(bizId, moduleId) {
    const b = bizRec(bizId), cat = version(b.v), mod = cat.modules.find((m) => m.id === moduleId);
    if (!mod) return { state: "on", inc: true };
    const cell = mod.cells[b.plan] || X();
    const r = { inc: !!cell.inc, off: cell.off || "locked", text: cell.text || "", limit: cell.limit, per: cell.per, name: mod.name, what: mod.what || "", override: "" };
    overrides(bizId).forEach((o) => {
      if (o.module !== moduleId) return;
      if (o.kind === "grant") { r.inc = true; r.text = "Granted"; r.limit = null; r.override = "Granted by Saroh"; }
      if (o.kind === "remove") { r.inc = false; r.off = "hidden"; r.override = "Removed by Saroh"; }
      if (o.kind === "limit") { r.limit = Number(o.value); r.text = Number(o.value).toLocaleString("en-IN") + (r.per ? " a " + r.per : ""); r.override = "Limit set by Saroh"; }
    });
    const bl = billing(bizId);
    (live().addons || []).forEach((a) => { const n = Number(bl.addons[a.id] || 0); if (!n) return;
      if (a.kind === "module" && a.module === moduleId && !r.inc) { r.inc = true; r.text = "Add-on"; r.limit = null; r.override = "Bought as an add-on"; }
      if (ADDON_MOD[a.kind] === moduleId && r.inc && r.limit != null) { r.limit = Number(r.limit) + n * (a.mode === "unit" ? 1 : Number(a.qty) || 1); r.override = r.override || "Raised by an add-on"; }
    });
    r.state = r.inc ? "on" : r.off;
    const plans = cat.plans.filter((p) => !p.retired), i = plans.findIndex((p) => p.id === b.plan);
    const up = plans.slice(Math.max(i + 1, 0)).find((p) => (mod.cells[p.id] || {}).inc);
    r.upgradeTo = up ? up.name : ""; r.upgradePrice = up ? up.price : 0;
    r.plan = (cat.plans.find((p) => p.id === b.plan) || {}).name || b.plan;
    return r;
  }
  function menuState(bizId, menu, child) {
    const b = bizRec(bizId), cat = version(b.v);
    const mod = cat.modules.find((m) => m.menu === menu && (child ? m.child === child : !m.child));
    return mod ? access(bizId, mod.id) : { state: "on" };
  }
  function price(bizId) {
    const b = bizRec(bizId), p = version(b.v).plans.find((x) => x.id === b.plan) || { price: 0 };
    const o = overrides(bizId).filter((x) => x.kind === "price").pop();
    return o ? Number(o.value) : p.price;
  }
  /* Card bullets: the first plan lists what it has; each later plan lists what's different from the one before. */
  function cardLines(cat, planId) {
    const plans = cat.plans.filter((p) => !p.retired), i = plans.findIndex((p) => p.id === planId), prev = plans[i - 1];
    const lines = [];
    cat.modules.forEach((m) => {
      if (m.pricing === "hidden") return;
      const c = m.cells[planId] || X(); if (!c.inc || !c.card) return;
      if (prev) { const pc = m.cells[prev.id] || X(); if (pc.inc && pc.text === c.text && pc.card === c.card) return; }
      lines.push({ t: c.card + (m.pricing === "soon" ? " (coming soon)" : ""), soon: m.pricing === "soon" });
    });
    return { lead: prev ? "Everything in " + prev.name + ", plus:" : "", lines: lines };
  }
  /* Limit notice shared by every screen: nothing under 80%, a warning from 80%, blocked at 100%. */
  function limitNotice(bizId, moduleId, count, what, pausedText) {
    const a = access(bizId, moduleId); if (!a.inc || a.limit == null) return { on: false, full: false };
    const L = Number(a.limit), n = Number(count) || 0, f = (x) => Number(x).toLocaleString("en-IN");
    if (n < 0.8 * L) return { on: false, full: false, left: L - n };
    const full = n >= L;
    return { on: true, full, left: Math.max(0, L - n), pct: Math.min(100, Math.round((n / L) * 100)) + "%",
      bar: full ? "var(--danger-text)" : "var(--accent)", line: full ? "var(--danger-line)" : "var(--accent-line)", bg: full ? "var(--danger-bg)" : "var(--accent-bg)",
      title: full ? "You've reached your " + f(L) + " " + what + " on " + a.plan : "You've used " + f(n) + " of " + f(L) + " " + what + " on " + a.plan,
      body: full ? pausedText + (a.upgradeTo ? " " + a.upgradeTo + " raises the limit, or add more with an add-on." : " Add more with an add-on.") : "You'll be stopped at " + f(L) + "." + (a.upgradeTo ? " " + a.upgradeTo + " gives you more." : " An add-on gives you more."),
      cta: a.upgradeTo ? "Upgrade or add more" : "Add more",
      why: full ? "You've reached your " + what + " limit on " + a.plan : "" };
  }
  window.SarohCatalog = { DEFAULT, limitNotice, clone, versions, live, liveV, scheduled, cancelScheduled, coupons, saveCoupons, billing, setBilling, todayISO, moveOnFrom: (iso) => fmtDate(iso ? billingFrom(new Date(iso + "T00:00:00")) : nextBilling()), version, draft, editing, saveDraft, discardDraft, publish, diff, nextBilling: () => fmtDate(nextBilling()),
    bizAll, bizRec, setBiz, onPlan, overrides, addOverride, removeOverride, access, menuState, price, cardLines,
    subscribe: (f) => { subs.add(f); return () => subs.delete(f); } };
})();
